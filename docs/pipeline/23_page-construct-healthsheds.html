Implement legacy stage 5 as ordered spatial targets

This page implements pinned legacy stage 5 at 31e0769683e133e136b09ac0c6608c5a12657a9a. Each reviewed correction round is a separate target. Shared FKT assignments, population accounting, missing-value behavior, join multiplicity, and residual non-contiguity retain legacy rules. Spatial handoffs are ordinary sf targets; nonspatial exports use Parquet. The execution page tangles these declarations without executing scientific code during rendering.

Tip Procedure: Declare Archive, Geometry, And Matching Helpers

Helpers are tracked function targets, not an unresolved source() call. The grouped fuzzy join preserves original row identifiers, shared-group boundaries, cosine distance, callback selection, and ties. The topology helper uses the legacy DE-9IM positions 1, 2, 4, and 5; point contact alone does not connect polygons. Relation matrices are decoded directly in R, avoiding a data-frame mutation for every clinic. Population summation likewise uses base-R pair deduplication: attaching duckplyr replaces unique.data.frame(), which would otherwise execute a DuckDB query for each clinic group. As in the legacy script, st_relate() uses planar DE-9IM on the source longitude/latitude coordinates; its repeated informational notice is suppressed, while errors and warnings propagate. Empty healthsheds are constructed with the input CRS.

s5_healthshed_runtime_fn <- function() {
  packages <- c("sf", "s2", "units", "igraph", "fuzzyjoin", "stringdist", "dplyr")

s5_healthshed_read_fn <- function(archive, members, name, kind = "csv", colClasses = NULL, ...) {
  s3_dictionary_archive_read_fn(archive, members[[name]], function(path) {
    if (kind == "xlsx") return(readxl::read_excel(path, ...))
    if (kind == "rds") return(readRDS(path))
    columns <- names(read.csv(path, nrows = 0L, fileEncoding = "UTF-8"))
    ids <- columns[grepl("uid|instat|code|pcode", columns, ignore.case = TRUE)]
    if (is.null(colClasses)) colClasses <- setNames(rep("character", length(ids)), ids)
    read.csv(path, colClasses = colClasses,
             fileEncoding = "UTF-8", stringsAsFactors = FALSE, ...)
  })
}
s5_healthshed_shapes_read_fn <- function(archive, members) {
  unsafe <- startsWith(members, "/") | vapply(strsplit(members, "/", fixed = TRUE), function(x) any(x == ".."), logical(1))
  stopifnot(length(members) >= 4L, !any(unsafe))
  required <- c("shp", "shx", "dbf", "prj")
  stopifnot(all(required %in% tools::file_ext(members)))
  scratch <- tempfile("healthshed-shapes-")
  dir.create(scratch)
  on.exit(unlink(scratch, recursive = TRUE), add = TRUE)
  paths <- utils::unzip(archive, files = members, exdir = scratch)
  stopifnot(all(file.exists(paths)))
  sf::read_sf(paths[endsWith(paths, ".shp")])
}
s5_healthshed_adjacency_fn <- function(x, y = x) {
  # st_relate uses planar DE-9IM even for geographic coordinates, as in stage 5.
  relation <- suppressMessages(sf::st_relate(x, y))
  matrix(grepl("1|2", paste0(substr(relation, 1, 2), substr(relation, 4, 5))),
         nrow = length(x), ncol = length(y))
}
s5_healthshed_components_fn <- function(x) {
  if (!length(x)) return(0L)
  igraph::components(igraph::graph_from_adjacency_matrix(s5_healthshed_adjacency_fn(x)), mode = "weak")$no
}
s5_grouped_stringdist_fulljoin_fn <- function(df1, df2, df1_group, df2_group,
    df1_stringmatch, df2_stringmatch, dist_method = "cosine", dist_col = "dist",
    max_dist = 30, ignore_case = TRUE, select_match = identity, unnest = TRUE, ...) {
  left <- as.data.frame(df1) %>% mutate(df1_id = seq_len(n())) %>% nest_by(across(all_of(df1_group)), .key = "df1")
  right <- as.data.frame(df2) %>% mutate(df2_id = seq_len(n())) %>% nest_by(across(all_of(df2_group)), .key = "df2")
  # Legacy full_join + removal of absent nested groups keeps shared groups only.
  joined <- inner_join(left, right, by = setNames(df2_group, df1_group)) %>%
    rowwise() %>% mutate(df_new = list(select_match(fuzzyjoin::stringdist_join(
      df1, df2, by = setNames(df2_stringmatch, df1_stringmatch), mode = "full",
      method = dist_method, distance_col = dist_col, max_dist = max_dist,
      ignore_case = ignore_case, ...))))
  if (unnest) joined <- joined %>% select(-df1, -df2) %>% tidyr::unnest(df_new)
  joined
}
s5_healthshed_refresh_fn <- function(x, lookup, shapes, fallback = FALSE) {
  x <- s5_healthshed_drop_geometry_fn(x)
  previous <- if (fallback && "ADM4_PCODE" %in% names(x)) x$ADM4_PCODE else rep(NA_character_, nrow(x))
  x$ADM4_PCODE <- NULL
  x$.s5_previous_pcode <- previous
  x <- left_join(x, select(lookup, ADM4C_BNGRC, ADM4_PCODE), by = c("fkt_instat_clean" = "ADM4C_BNGRC"))
  if (fallback) x$ADM4_PCODE <- coalesce(x$ADM4_PCODE, x$.s5_previous_pcode)
  x$.s5_previous_pcode <- NULL
  x <- left_join(x, as.data.frame(select(shapes, ADM4_PCODE)), by = "ADM4_PCODE")
  sf::st_as_sf(x)
}

                  sf::st_combine() %>%
                  sf::st_make_valid()} else {empty_geom}}) %>%
    rowwise() %>%
    mutate(n_shape = length(geometry[[1]])) %>%
    ungroup
}
