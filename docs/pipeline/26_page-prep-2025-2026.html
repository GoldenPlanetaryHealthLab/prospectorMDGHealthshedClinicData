The loader above evaluates function definitions only. Target declarations are reserved for the graph. User-run examples use plain R fences. The source resolver follows Stagecoach’s input directory/source/item name/source basename layout. It rebases the manifest’s project-local input directory through here::here() so host absolute paths do not leak into the container. The two source patterns are explicit directory paths, not discovery globs.

s8_new_wave_roots_fn <- function(manifest_path = here::here("stagecoach_manifest.yml")) {
  manifest <- yaml::read_yaml(manifest_path, eval.expr = FALSE)
  source <- manifest$sources[["01_gold_mine"]]
  if (!isTRUE(source$enabled)) stop("Stagecoach's 01_gold_mine source is disabled.")
  project <- sub("/+$", "", manifest$project$project_working_dir)
  input <- manifest$project$input_data_dir
  if (length(project) != 1L || length(input) != 1L || !startsWith(input, paste0(project, "/"))) {
    stop("The Stagecoach input directory must be inside its declared project directory.")
  }
  input_relative <- substring(input, nchar(project) + 2L)
  ids <- c("harvard_2025_2026", "july_2025_candidate")
  items <- lapply(ids, function(id) {
    hits <- Filter(function(x) identical(x$name, id), source$items)
    if (length(hits) != 1L) stop("Expected one Stagecoach item named ", id)
    pattern <- hits[[1L]]$path_regex
    if (length(pattern) != 1L || !nzchar(pattern) ||
        any(vapply(c("*", "?", "["), function(x) grepl(x, pattern, fixed = TRUE), TRUE))) {
      stop("New-wave Stagecoach items must name explicit source directories: ", id)
    }
    hits[[1L]]
  })
  tibble::tibble(
    source_id = ids,
    root = vapply(items, function(item) {
      here::here(input_relative, "01_gold_mine", item$name, basename(item$path_regex))
    }, ""),
    scope = c("approved Harvard delivery", "approved separate July delivery")
  )
}

s8_new_wave_inventory_fn <- function(roots, hash = FALSE) {
  stopifnot(nrow(roots) > 0L, !anyDuplicated(roots$source_id))
  dplyr::bind_rows(lapply(seq_len(nrow(roots)), function(i) {
    root <- roots$root[[i]]
    if (!dir.exists(root)) stop("Delivery root is unavailable: ", root)
    relative <- sort(list.files(root, recursive = TRUE, full.names = FALSE,
                                include.dirs = FALSE), method = "radix")
    paths <- file.path(root, relative)
    info <- file.info(paths)
    parts <- strsplit(relative, "/", fixed = TRUE)
    tibble::tibble(
      source_id = roots$source_id[[i]], relative_path = relative, path = paths,
      delivery = vapply(parts, function(x) if (length(x) >= 3L) x[[2L]] else roots$source_id[[i]], ""),
      extension = tolower(tools::file_ext(paths)), bytes = info$size,
      modified = info$mtime,
      content_md5 = if (hash) unname(tools::md5sum(paths)) else rep(NA_character_, length(paths))
    )
  }))
}

s8_new_wave_metadata_fn <- function() {
  c(paste0("orgunitlevel", 1:5), "organisationunitid", "organisationunitname",
    "organisationunitcode", "organisationunitdescription", "periodid",
    "periodname", "periodcode", "perioddescription")
}

s8_new_wave_header_fn <- function(path) {
  columns <- names(readr::read_csv(
    path, n_max = 0L, col_types = readr::cols(.default = readr::col_character()),
    name_repair = "minimal", show_col_types = FALSE, progress = FALSE
  ))
  if (anyNA(columns) || any(!nzchar(columns)) || anyDuplicated(columns)) {
    stop("Review empty or repeated source headers before reading: ", path)
  }
  columns
}
Tip Procedure: Read Raw Text Through DuckDB

Keep all source fields as VARCHAR initially. Preserve identifier spelling, leading zeros, and measurement tokens for the conversion audit on page 27. Empty fields are null; literal NA remains text at this raw boundary. This is an intentional difference from the already-curated numeric reader on page 18. No coercion, renaming, grouping, row filtering, or automatic error skipping occurs here. Header checks fail before DuckDB can silently repair duplicates.

s8_new_wave_literal_fn <- function(x) {
  paste0("'", gsub("'", "''", x, fixed = TRUE), "'")
}
s8_new_wave_identifier_fn <- function(x) {
  paste0('"', gsub('"', '""', x, fixed = TRUE), '"')
}
s8_new_wave_csv_sql_fn <- function(path) {
  stopifnot(length(path) == 1L, file.exists(path))
  paste0("read_csv(", s8_new_wave_literal_fn(path),
         ", header=true, delim=',', quote='\"', escape='\"', ",
         "all_varchar=true, nullstr='', strict_mode=true, ignore_errors=false)")
}
s8_new_wave_read_fn <- function(path) {
  original <- s8_new_wave_header_fn(path)
  relation <- duckplyr::read_sql_duckdb(
    paste0("SELECT * FROM ", s8_new_wave_csv_sql_fn(path)), prudence = "stingy"
  )
  if (!identical(names(relation), original)) stop("DuckDB changed the source header: ", path)
  relation
}

s8_new_wave_shape_fn <- function(path) {
  relation <- s8_new_wave_read_fn(path)
  required <- c("organisationunitid", "periodcode", "periodname")
  missing <- setdiff(required, names(relation))
  if (length(missing)) stop("Missing survey keys in ", path, ": ", paste(missing, collapse = ", "))
  source <- s8_new_wave_csv_sql_fn(path)
  query <- paste0(
    "SELECT count(*) AS rows, count(DISTINCT organisationunitid) FILTER ",
    "(WHERE organisationunitid IS NOT NULL AND trim(organisationunitid) NOT IN ('','NA')) AS clinics, ",
    "count(*) FILTER (WHERE organisationunitid IS NULL OR trim(organisationunitid) IN ('','NA')) AS missing_uid, ",
    "count(*) FILTER (WHERE periodcode IS NULL OR trim(periodcode) IN ('','NA')) AS missing_period, ",
    "count(*) FILTER (WHERE periodcode IS NOT NULL AND trim(periodcode) NOT IN ('','NA') ",
    "AND NOT regexp_full_match(periodcode, '[0-9]{4}(0[1-9]|1[0-2])')) AS invalid_period, ",
    "(SELECT count(*) FROM (SELECT organisationunitid,periodcode FROM ", source,
    " GROUP BY organisationunitid,periodcode)) AS clinic_month_keys FROM ", source
  )
  duckplyr::read_sql_duckdb(query, prudence = "stingy") |>
    dplyr::collect() |>
    dplyr::mutate(columns = length(names(relation)),
                  measurement_columns = length(setdiff(names(relation), s8_new_wave_metadata_fn())))
}
Tip Procedure: Write The Raw Parquet Boundary

The writer is called by one file-target branch. It records a content hash and checks it again after the read, preserving every source column and adding provenance fields. Targets tracks each input CSV directly. The source-row field is a read-time record ordinal, not a physical CSV line number or a stable clinic identifier. These raw Parquet products retain the observed source snapshot. They do not replace the staged source CSVs or the old-wave baseline.

s8_new_wave_write_raw_fn <- function(path, output_path, source_id, relative_path, expected_md5 = NULL) {
  if (is.null(expected_md5)) expected_md5 <- unname(tools::md5sum(path))
  stopifnot(length(expected_md5) == 1L, !is.na(expected_md5), nzchar(expected_md5),
            length(source_id) == 1L, length(relative_path) == 1L,
            identical(unname(tools::md5sum(path)), expected_md5))
  relation <- s8_new_wave_read_fn(path)
  reserved <- c("_source_file", "_source_row", "_source_id", "_source_md5")
  if (any(reserved %in% names(relation))) stop("Source uses reserved provenance column names.")
  sql <- paste0("SELECT *, ", s8_new_wave_literal_fn(relative_path), " AS _source_file, ",
                "row_number() OVER () AS _source_row, ", s8_new_wave_literal_fn(source_id),
                " AS _source_id, ", s8_new_wave_literal_fn(expected_md5), " AS _source_md5 FROM ",
                s8_new_wave_csv_sql_fn(path))
  dir.create(dirname(output_path), recursive = TRUE, showWarnings = FALSE)
  duckplyr::compute_parquet(
    duckplyr::read_sql_duckdb(sql, prudence = "stingy"), output_path
  )
  if (!identical(unname(tools::md5sum(path)), expected_md5)) stop("Source changed during ingestion.")
  output_path
}

The execution page branches over the Stagecoach-derived file inventory, then tracks each CSV and its raw Parquet separately.

Tip Procedure: Inventory Stagecoach Sources And Describe Each File

The execution page refreshes the list of staged CSV paths before each build. Individual file targets detect content changes. The source snapshot below records the hashes captured by ingestion as derived provenance; it is never read back as configuration or as an input-discovery prerequisite.

s8_new_wave_sources_fn <- function(roots) {
  x <- s8_new_wave_inventory_fn(roots, hash = FALSE)
  x <- x[x$extension == "csv", c("source_id", "relative_path", "path")]
  if (!nrow(x)) stop("No new-wave CSVs are staged through Stagecoach.")
  if (anyDuplicated(x[c("source_id", "relative_path")])) stop("Repeated staged CSV paths.")
  x$source_file <- paste(x$source_id, x$relative_path, sep = "/")
  x
}
s8_new_wave_snapshot_fn <- function(paths, inventory) {
  hashes <- dplyr::bind_rows(lapply(paths, function(path) {
    duckplyr::read_parquet_duckdb(path, prudence = "stingy") |>
      dplyr::distinct(`_source_id`, `_source_file`, `_source_md5`) |>
      dplyr::collect() |>
      dplyr::rename(source_id = `_source_id`, source_file = `_source_file`, content_md5 = `_source_md5`)
  }))
  stopifnot(nrow(hashes) == nrow(inventory), !anyDuplicated(hashes[c("source_id", "source_file")]))
  snapshot <- dplyr::left_join(dplyr::select(inventory, source_id, relative_path, source_file), hashes,
    by = c("source_id", "source_file"), relationship = "one-to-one")
  stopifnot(!anyNA(snapshot$content_md5))
  snapshot
}
s8_new_wave_table_key_fn <- function(path) {
  parts <- strsplit(tolower(path), "/", fixed = TRUE)[[1L]]
  table <- which(grepl("^tableau ", parts))
  key <- paste(parts[seq.int(if (length(table)) tail(table, 1L) else length(parts), length(parts))], collapse = "/")
  key <- gsub("janv\\s*-\\s*juin|juillet\\s*-\\s*dec|\\b[0-9]{6}\\b|\\b20[0-9]{2}\\b", " ", key, perl = TRUE)
  trimws(gsub(" +", " ", gsub(" .csv", ".csv", key, fixed = TRUE)))
}
s8_new_wave_schema_fn <- function(paths) {
  dplyr::bind_rows(lapply(paths, function(path) {
    relation <- duckplyr::read_parquet_duckdb(path, prudence = "stingy")
    info <- relation |> dplyr::distinct(`_source_id`, `_source_file`) |> dplyr::collect()
    stopifnot(nrow(info) == 1L)
    periods <- relation |> dplyr::distinct(periodcode) |> dplyr::collect()
    cols <- setdiff(names(relation), c("_source_id", "_source_file", "_source_row", "_source_md5"))
    tibble::tibble(source_id = info$`_source_id`, source_file = info$`_source_file`,
      table_key = s8_new_wave_table_key_fn(info$`_source_file`), position = seq_along(cols),
      source_name = cols, is_metadata = cols %in% s8_new_wave_metadata_fn(),
      periods = rep(list(sort(unique(periods$periodcode))), length(cols)))
  }))
}
