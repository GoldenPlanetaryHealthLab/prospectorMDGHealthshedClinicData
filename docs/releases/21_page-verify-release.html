31  Verify an Internal Candidate

Check the packaged files independently of successful target execution

Verification reopens each candidate resource, compares its schema and complete row multiset with its curated source, and records checksums and row counts. It verifies serialization, not scientific equivalence with an archived release.

Important Prerequisite: Require The Immutable Resource Contract

Package copies must refer to the exact source hashes in the versioned contract. A changed source or modified package must fail verification. The default access remains internal and the release status remains candidate.

Tip Procedure: Verify Every Resource And Record Its Schema 
srelease_verify_fn <- function(resource,package,contract,config) {
  manifest <- duckplyr::read_parquet_duckdb(contract,prudence="stingy") |> dplyr::collect()
  row <- manifest[manifest$resource==resource$name,]
  stopifnot(nrow(row)==1L,identical(srelease_sha256_fn(package),row$source_sha256),
    identical(srelease_sha256_fn(resource$source),row$source_sha256))
  a <- paste("SELECT * FROM",s4a_diagnosis_read_fn(resource$source))
  b <- paste("SELECT * FROM",s4a_diagnosis_read_fn(package))
  schema_a <- s4a_diagnosis_query_fn(paste("DESCRIBE",a)) |> dplyr::collect()
  schema_b <- s4a_diagnosis_query_fn(paste("DESCRIBE",b)) |> dplyr::collect()
  differences <- schecks_key_difference_fn(a,b)
  if(!identical(schema_a$column_name,schema_b$column_name) || !identical(schema_a$column_type,schema_b$column_type) || any(differences!=0)) stop("Packaged resource does not round-trip to its curated source.")
  if(any(schema_b$column_name %in% config$prohibited_columns)) stop("Prohibited column in packaged resource.")
  rows <- s4a_diagnosis_query_fn(paste0("SELECT count(*) AS n FROM (",b,")")) |> dplyr::collect()
  tibble::tibble(resource=resource$name,rows=rows$n,columns=nrow(schema_b),sha256=row$source_sha256,
    schema_equal=TRUE,rows_equal=TRUE,access=config$access,status=config$status)
}
srelease_schema_fn <- function(paths,root,path) {
  table <- dplyr::bind_rows(lapply(paths,function(input) {
    schema <- s4a_diagnosis_query_fn(paste("DESCRIBE SELECT * FROM",s4a_diagnosis_read_fn(input))) |> dplyr::collect()
    tibble::tibble(resource=substring(input,nchar(root)+2L),position=seq_len(nrow(schema)),column=schema$column_name,type=schema$column_type)
  }))
  srelease_write_once_fn(table,path)
}
srelease_manifest_fn <- function(checks,resources,config,path) {
  stopifnot(nrow(checks)==length(resources),!anyDuplicated(checks$resource),all(checks$schema_equal & checks$rows_equal))
  stopifnot(all(checks$access==config$access),all(checks$status==config$status))
  if(file.exists(path)) {
    old <- duckplyr::read_parquet_duckdb(path,prudence="stingy") |> dplyr::collect() |> dplyr::arrange(resource)
    if(!isTRUE(all.equal(as.data.frame(old),as.data.frame(dplyr::arrange(checks,resource)),check.attributes=FALSE))) stop("Immutable verification manifest differs.")
    return(path)
  }
  s5_healthshed_write_table_fn(dplyr::arrange(checks,resource),path)
}
